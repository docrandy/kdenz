    ```typescript
     * SelfAssessment Component

      focusMode: 'filler' | 'pace';
      onComplete: (response: SelfAssessmentResponse) => void;
      onSkip: () => void;
    }

    export interface SelfAssessmentResponse {
      rating: 1 | 2 | 3 | 4 | 5;  // 1=struggled, 5=great
      note?: string;  // Optional brief note
    }

    export default function SelfAssessment({ focusMode, onComplete, onSkip }: SelfAssessmentProps) {
      // State for rating selection
      // Question varies by focus mode:
      // - Filler: "Before we show your results, how aware were you of your filler words?"
      // - Pace: "Before we show your results, how did your speaking pace feel?"

      // 5-point scale with neutral labels (no judgment):
      // 1: "Not very aware" / "Felt rushed"
      // 2: "Somewhat aware" / "A bit fast"
      // 3: "Moderately aware" / "About right"
      // 4: "Quite aware" / "Comfortable"
      // 5: "Very aware" / "Felt controlled"

      // Optional: "Any thoughts about this session?" (single line input, not required)

      // Buttons: "Show My Results" (primary), "Skip" (text link)
      // GAIN-aligned: No pressure, user controls when to see data
    }
    ```

    Add state machine for reflection flow:
    1. Show SelfAssessment first (before any metrics)
    2. Show metrics/results after self-assessment complete or skipped
    3. Show ImplementationIntention after user has seen metrics
    4. Final state with all navigation buttons

    ```typescript
    // Add imports
    import SelfAssessment, { SelfAssessmentResponse } from '../components/SelfAssessment';
    import ImplementationIntention from '../components/ImplementationIntention';

    // Add state for flow
    type ResultsPhase = 'self-assess' | 'metrics' | 'intention' | 'complete';
    const [phase, setPhase] = useState<ResultsPhase>('self-assess');
    const [selfAssessment, setSelfAssessment] = useState<SelfAssessmentResponse | null>(null);

    // Handlers
    const handleSelfAssessComplete = (response: SelfAssessmentResponse) => {
      setSelfAssessment(response);
      setPhase('metrics');
    };

    const handleSelfAssessSkip = () => {
      setPhase('metrics');
    };

    const handleMetricsContinue = () => {
      setPhase('intention');
    };

    const handleIntentionComplete = (intention: string) => {
      // Optionally store intention
      sessionStorage.setItem('voicelab_last_intention', intention);
      setPhase('complete');
    };

    const handleIntentionSkip = () => {
      setPhase('complete');
    };
    ```

    Wrap existing content in conditional rendering based on phase:

    ```tsx
    // In JSX return:
    {phase === 'self-assess' && (
      <SelfAssessment
        focusMode={sessionData.focusMode}
        onComplete={handleSelfAssessComplete}
        onSkip={handleSelfAssessSkip}
      />
    )}

    {phase === 'metrics' && (
      <>
        {/* Existing metrics/scorecard display */}
        <div>
          {/* Current session metrics here */}
        </div>
        <button onClick={handleMetricsContinue}>Continue</button>
      </>

    {phase === 'intention' && (

        focusMode={sessionData.focusMode}

      />
    )}

    {phase === 'complete' && (
      <>
        {/* Full results view with navigation buttons */}
        {/* "Practice Again" / "End Session" buttons */}
      </>
    )}
    ```
