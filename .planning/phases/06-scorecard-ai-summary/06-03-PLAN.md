    ```typescript
    import { saveSession } from '../services/sessionStorage';
    import WeeklyTrendChart from '../components/WeeklyTrendChart';

    // Add state to track if session was saved
    const [sessionSaved, setSessionSaved] = useState(false);
    const [refreshKey, setRefreshKey] = useState(0);

    // Save session on mount (in useEffect, after loading sessionData)
    useEffect(() => {
      if (sessionData && !sessionSaved && !sessionData.is_baseline) {
        saveSession({
          durationSeconds: sessionData.durationSeconds,
          wordCount: sessionData.wordCount,
          wpm: sessionData.wpm,
          fillerCount: sessionData.fillerCount,
          fillerRate: sessionData.fillerRate,
          focusMode: sessionData.focusMode,
        });
        setSessionSaved(true);
        setRefreshKey(k => k + 1); // Trigger chart refresh
      }
    }, [sessionData, sessionSaved]);

    // In render (after Scorecard, before navigation):
    {!sessionData.is_baseline && (
      <div className="mt-6">
        <WeeklyTrendChart refreshKey={refreshKey} />
      </div>
    )}
    ```

    Note: Don't save baseline sessions to regular session storage (they go to baselineStorage)
    Note: refreshKey forces chart to re-render after new session is saved
