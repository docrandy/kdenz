    ```typescript
    interface InstituteProgress {
      consumed: string[];     // content IDs user has completed
      prescribed: string[];   // content IDs system has recommended
      dismissed: string[];    // content IDs user dismissed
    }

    export function getInstituteProgress(): InstituteProgress {
      try {
        const raw = localStorage.getItem(PROGRESS_KEY);
        if (!raw) return { consumed: [], prescribed: [], dismissed: [] };
        return JSON.parse(raw) as InstituteProgress;
      } catch {
        return { consumed: [], prescribed: [], dismissed: [] };
      }
    }

    export function markContentConsumed(contentId: string): void {
      try {
        const progress = getInstituteProgress();
        if (!progress.consumed.includes(contentId)) {
    ```

    ```typescript
    import type { LearnerProfile } from '../types/institute';
    import type { PatternType } from '../types/simulation';
    import type { ArchetypeId } from '../types/aspiration';
    import { getAspiration } from './aspirationStorage';
    import { getInstituteProgress } from './instituteStorage';

    const PATTERN_HISTORY_KEY = 'kdenz:pattern-history';
    const MAX_PATTERN_HISTORY = 10;

    export function getLearnerProfile(): LearnerProfile {
      const aspiration = getAspiration();
      const progress = getInstituteProgress();
      const patternHistory = getPatternHistory();

      // Determine current pattern from most recent entry
      const current_pattern = patternHistory.length > 0
        ? patternHistory[patternHistory.length - 1]
        : null;

      // Confidence based on consistency in history
      let pattern_confidence: 'low' | 'medium' | 'high' = 'low';
      if (patternHistory.length >= 5) {
        const modeCount = patternHistory.filter(p => p === current_pattern).length;
        const ratio = modeCount / patternHistory.length;
        if (ratio >= 0.7) pattern_confidence = 'high';
        else if (ratio >= 0.4) pattern_confidence = 'medium';
      } else if (patternHistory.length >= 3) {
        pattern_confidence = 'medium';
      }

      return {
        current_pattern,
        pattern_history: patternHistory,
        pattern_confidence,
        stated_aspiration: aspiration?.primary_archetype_id ?? null,
        mastered_technique_ids: [],   // populated by mastery tracking (Phase 21)
        attempted_technique_ids: [],  // populated by drill attempt storage
        skill_gaps: [],               // populated by mastery tracking (Phase 21)
        consumed_content_ids: progress.consumed,
        prescribed_content_ids: progress.prescribed,
        dismissed_content_ids: progress.dismissed,
        vcm_gate_failures: [],        // stub — empty until Phase D
      };
    }

    export function addPatternToHistory(pattern: PatternType): void {
      try {
        const history = getPatternHistory();
        history.push(pattern);
        // Rolling window of MAX_PATTERN_HISTORY
        const trimmed = history.slice(-MAX_PATTERN_HISTORY);
        localStorage.setItem(PATTERN_HISTORY_KEY, JSON.stringify(trimmed));
      } catch {
        console.warn('[LearnerProfile] Failed to save pattern history');
      }
    }

    function getPatternHistory(): PatternType[] {
      try {
        const raw = localStorage.getItem(PATTERN_HISTORY_KEY);
        if (!raw) return [];
        return JSON.parse(raw) as PatternType[];
      } catch {
        return [];
      }
    }
    ```

    ```typescript
    import type { InstituteContentItem, LearnerProfile } from '../types/institute';
    import { INSTITUTE_CONTENT } from '../data/instituteContent';

    export interface ScoredContent {
      item: InstituteContentItem;
      score: number;
      reason: string;  // "why this for you" personalization
    }

    export function scoreContentItem(
      item: InstituteContentItem,
      profile: LearnerProfile,
    ): ScoredContent {
      let score = 0;
      const reasons: string[] = [];

      // Pattern match (50 points max)
      if (profile.current_pattern && item.addresses_patterns.includes(profile.current_pattern)) {
        score += 50;
        reasons.push(`Addresses your ${profile.current_pattern.replace('-', ' ')} pattern`);
      }

      // Aspiration match (30 points max)
      if (profile.stated_aspiration && item.supports_aspirations.includes(profile.stated_aspiration)) {
        score += 30;
        reasons.push(`Builds toward your target voice`);
      }

      // Skill gap match (20 points max)
      const gapOverlap = item.technique_ids.filter(t => profile.skill_gaps.includes(t));
      if (gapOverlap.length > 0) {
        score += 20;
        reasons.push(`Covers techniques you haven't mastered`);
      }

      // VCM gate match (10 points max) — stub, uses empty array for now
      const gateOverlap = item.vcm_gates.filter(g => profile.vcm_gate_failures.includes(g));
      if (gateOverlap.length > 0) {
        score += 10;
      }

      // Penalties
      // Already consumed: -100 (filter out)
      if (profile.consumed_content_ids.includes(item.id)) {
        score -= 100;
      }
      // Dismissed: -100 (filter out)
      if (profile.dismissed_content_ids.includes(item.id)) {
        score -= 100;
      }
      // Prerequisites not met: -50
      const unmetPrereqs = item.prerequisite_ids.filter(
        prereq => !profile.consumed_content_ids.includes(prereq)
      );
      if (unmetPrereqs.length > 0) {
        score -= 50;
      }

      return {
        item,
        score,
        reason: reasons.length > 0 ? reasons[0] : 'Recommended for your level',
      };
    }

    /**
     * Get top N content items ranked by relevance to learner profile.
     * debriefOnly: true returns only debrief_eligible items (for post-session surface)
     */
    export function getTopContent(
      profile: LearnerProfile,
      count: number = 2,
      debriefOnly: boolean = false,
    ): ScoredContent[] {
      let items = INSTITUTE_CONTENT;
      if (debriefOnly) {
        items = items.filter(item => item.debrief_eligible);
      }

      const scored = items
        .map(item => scoreContentItem(item, profile))
        .filter(s => s.score > 0)  // exclude negative scores (consumed, dismissed)
        .sort((a, b) => {
          // Primary: score descending
          if (b.score !== a.score) return b.score - a.score;
          // Secondary: tier ascending (simpler content first)
          return a.item.tier - b.item.tier;
        });

      return scored.slice(0, count);
    }

    /**
     * Get personalized debrief hook text for a content item.
     * Replaces {count} and {avgWordCount} placeholders with session data.
     */
    export function formatDebriefHook(
      hook: string,
      sessionData: { patternCount?: number; avgWordCount?: number },
    ): string {
      let formatted = hook;
      if (sessionData.patternCount !== undefined) {
        formatted = formatted.replace('{count}', String(sessionData.patternCount));
      }
      if (sessionData.avgWordCount !== undefined) {
        formatted = formatted.replace('{avgWordCount}', String(sessionData.avgWordCount));
      }
      return formatted;
    }
    ```
