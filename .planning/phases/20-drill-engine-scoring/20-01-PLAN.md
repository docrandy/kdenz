    export type DrillEvaluationResult = {
      formScore: number;         // 0-100, computed instantly
      accuracyScore: number | null;    // 0-100 from Gemini, null if pending/failed
      impactScore: number | null;      // 0-100 from Gemini, null if pending/failed
      timingScore: number;       // always 0 for Phase 20 isolated drills
      compositeScore: number;    // weighted: form*0.25 + accuracy*0.35 + impact*0.30 + timing*0.10
                                 // if accuracy/impact null: compute composite on form alone (scale it)
      explanation: string;       // coaching narrative from Gemini, or fallback string if Gemini fails
      geminiSucceeded: boolean;  // false if Gemini failed or no API key
    }

    export function scoreForm(transcript: string, technique: Technique): number
      - Apply each SyntaxRule from technique.syntax_rules to the transcript
      - For 'regex' rules: test new RegExp(rule.pattern, 'i').test(transcript) — pass = contributes weight
      - For 'negation' rules: pass = pattern NOT found in transcript — contributes weight
      - For 'inclusion' rules: test new RegExp(rule.pattern, 'i').test(transcript) — same as regex
      - Collect: totalWeight = sum of all rule.weight values (default 1.0 if undefined)
      - Collect: passedWeight = sum of weights for rules that passed
      - Return: Math.round((passedWeight / totalWeight) * 100)
      - Edge case: if transcript is empty or whitespace-only, return 0
      - Edge case: if technique.syntax_rules is empty, return 100

    export async function scoreWithGemini(
      transcript: string,
      technique: Technique,
      scenario: Scenario,
      apiKey: string | null
    ): Promise<{ accuracyScore: number | null; impactScore: number | null; explanation: string; succeeded: boolean }>
      - If apiKey is null: return { accuracyScore: null, impactScore: null, explanation: "Accuracy and Impact scoring requires a Gemini API key.", succeeded: false }
      - Build prompt (see below)
      - Fetch to 'https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${apiKey}'
        with method POST, Content-Type application/json
      - Body: { contents: [{ parts: [{ text: prompt }] }], generationConfig: { temperature: 0.3, maxOutputTokens: 300 } }
      - Parse response: extract candidates[0].content.parts[0].text
      - Extract JSON block from text: look for ```json ... ``` or just the raw JSON object
      - Parse JSON: expect { accuracy: number, impact: number, explanation: string }
      - Clamp accuracy and impact to 0-100
      - Return { accuracyScore: accuracy, impactScore: impact, explanation, succeeded: true }
      - On any error (fetch fail, timeout, parse fail, non-ok response): return { accuracyScore: null, impactScore: null, explanation: "Accuracy and Impact scoring is temporarily unavailable.", succeeded: false }
      - Wrap entire function in try/catch

    GEMINI PROMPT for scoreWithGemini:
    ```
    You are evaluating a communication technique drill response.

    TECHNIQUE: ${technique.name}
    DESCRIPTION: ${technique.description}
    SYNTAX TEMPLATE: ${technique.syntax_template}

    SCENARIO PROMPT:
    ${scenario.prompt}

    MODEL ANSWER (reference, not required to match exactly):
    ${scenario.model_answer}

    USER'S RESPONSE:
    "${transcript}"

    ${scenario.evaluation_notes ? `EVALUATOR NOTES: ${scenario.evaluation_notes}\n\n` : ''}Rate the response on two dimensions (0-100 each):

    ACCURACY (0-100): How well does the response apply the core technique? Does it match the intent of the model answer? Does it address what the scenario requires?

    IMPACT (0-100): How effective would this response be on the counterpart in a real conversation? Would it create the intended effect (connection, open dialogue, de-escalation, etc.)?

    Respond with ONLY valid JSON in this exact format:
    {"accuracy": <number>, "impact": <number>, "explanation": "<3-4 sentence coaching narrative>"}

    The explanation should address what worked and one concrete improvement. Be specific.
    ```

    export async function evaluateDrillResponse(
      transcript: string,
      technique: Technique,
      scenario: Scenario,
      apiKey: string | null
    ): Promise<DrillEvaluationResult>
      - Run scoreForm synchronously
      - Run scoreWithGemini asynchronously (in same function body — await it)
      - Compute compositeScore:
          if accuracyScore !== null and impactScore !== null:
            compositeScore = Math.round(formScore * 0.25 + accuracyScore * 0.35 + impactScore * 0.30 + 0 * 0.10)
          else (Gemini failed — form only):
            compositeScore = formScore  // show form score as composite when LLM unavailable
      - Return DrillEvaluationResult with all fields
